function connectWS() { try { // Using a reliable public echo/relay service socket = new WebSocket('wss://free.relay.sockets.io'); socket.onopen = () => { const status = document.getElementById('connectionStatus'); status.textContent = 'SYSTEM ONLINE - STATION LINK ACTIVE'; status.className = 'status-online'; }; socket.onclose = () => { const status = document.getElementById('connectionStatus'); status.textContent = 'SYSTEM OFFLINE - RECONNECTING...'; status.className = 'status-offline'; setTimeout(connectWS, 2000); }; socket.onerror = () => { // Fallback local connection indicator if external WS fails const status = document.getElementById('connectionStatus'); status.textContent = 'LOCAL STANDALONE MODE ACTIVE'; status.className = 'status-online'; }; socket.onmessage = (e) => { try { const data = JSON.parse(e.data); if (data.type === 'SABOTAGE') { triggerSabotage(data.target); } else if (data.type === 'SYNC_ROBOT') { if (data.role === 'CRAWLER') stationData.CRAWLER.stealthPwr = data.state.stealthPwr; if (data.role === 'DRAINER') { stationData.DRAINER.battery = data.state.battery; if (data.state.stealPwr !== undefined) stationData.DRAINER.stealPwr = data.state.stealPwr; } if (data.role === 'TANK') stationData.TANK.immunityPwr = data.state.isImmune ? 0 : stationData.TANK.immunityPwr; updateUI(); } } catch (err) {} }; } catch(e) {} }
